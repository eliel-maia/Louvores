📋 Ver script SQL para criar as tabelas no Supabase

Execute no SQL Editor do seu painel Supabase:

CREATE TABLE IF NOT EXISTS louvores (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  titulo TEXT NOT NULL,
  artista TEXT NOT NULL,
  tonalidade TEXT NOT NULL,
  letra TEXT,
  cifra TEXT,
  youtube TEXT,
  cadastro TEXT DEFAULT to_char(now(), 'DD/MM/YYYY'),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS terca_louvores (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  louvor_id UUID REFERENCES louvores(id) ON DELETE CASCADE,
  ordem INTEGER NOT NULL DEFAULT 0,
  data DATE NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS domingo_louvores (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  louvor_id UUID REFERENCES louvores(id) ON DELETE CASCADE,
  ordem INTEGER NOT NULL DEFAULT 0,
  data DATE NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS louvor_historico (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  louvor_id UUID REFERENCES louvores(id) ON DELETE CASCADE,
  tab TEXT NOT NULL,
  data_uso DATE NOT NULL,
  deleted_at TIMESTAMP WITH TIME ZONE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

ALTER TABLE louvores ENABLE ROW LEVEL SECURITY;
ALTER TABLE terca_louvores ENABLE ROW LEVEL SECURITY;
ALTER TABLE domingo_louvores ENABLE ROW LEVEL SECURITY;
ALTER TABLE louvor_historico ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Acesso publico louvores" ON louvores FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acesso publico terca" ON terca_louvores FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acesso publico domingo" ON domingo_louvores FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acesso publico historico" ON louvor_historico FOR ALL USING (true) WITH CHECK (true);
